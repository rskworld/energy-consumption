About This Dataset

Usage Example

import pandas as pd

# Load the dataset
df = pd.read_csv('energy_consumption.csv')

# Basic statistics
print(df.describe())

# Peak hour analysis
hourly_avg = df.groupby('hour')['consumption_kwh'].mean()
print(hourly_avg.sort_values(ascending=False).head())
